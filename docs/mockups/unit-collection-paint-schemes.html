Reusable widget: Section

One widget gives Unit, Collection and Paint Scheme screens the same section boundary. Widget tests can find each header by its title, in order.

/// A titled block of a screen: amber bar + stencil caps title, optional count,
/// trailing actions, then [child] in a filled card unless [card] is false.
class Section extends StatelessWidget {
  const Section({super.key, required this.title, this.count,
      this.actions = const [], required this.child, this.card = true});
  final String title; final int? count;
  final List<Widget> actions;          // IconButtons, amber, 48dp
  final Widget child; final bool card;
  // build: Padding(h:16, bottom:20) > Column[
  //   SizedBox(height:48, Row[Container(4x18, primary), Text(title.toUpperCase(),
  //     style: stencil(titleSmall)), if count Text(count), Spacer(), ...actions]),
  //   card ? Card.filled(margin: zero, clipBehavior: antiAlias,
  //            color: surfaceContainerLow, child: child) : child ]
}
