Proposed shared pieces

/// The header row of Section, extracted so flat lists (Collections, Units) use it too.
/// Collapsible when expanded is non-null: whole row taps to toggle, chevron last.
SectionHeader({
  required String title,
  int? count,
  List<Widget> actions = const [],   // accent IconButtons, before the chevron
  bool? expanded,                       // null = not collapsible
  VoidCallback? onToggle,
  String? summary,                      // one muted line under the title while collapsed
  bool muted = false,                   // Uncategorised: outline bar, muted title
})
Section(...) keeps its API and builds SectionHeader.

/// Per-device collapse state, like ThemeSetting: a ChangeNotifier over collapsed.txt (one key per line).
CollapsedGroups.load(File) / isCollapsed(String key) / toggle(String key)
keys: 'game-system:' + lower-cased name ('' = Other), 'category:' + id, 'uncategorised:' + collectionId

/// The Category picker in UnitDialog (add and edit).
CategoryField({required List<Category> categories, String? categoryId, bool enabled = true,
  required ValueChanged<CategoryChoice> onChanged})   // existing id, null (Uncategorised) or a new name
